# ---------------------------------------------------------
# 1. 基础遥控测试 (无摄像头)

# ---------------------------------------------------------
# 2. 特殊串口 + 单摄像头测试
# 适用于使用 CH343 等国产串口芯片的情况 (ttyCH343USB*)
# 只启用前置摄像头 (index 0)
# ---------------------------------------------------------
lerobot-teleoperate \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyCH343USB1 \
    --robot.id=kang_follower_arm \
    --robot.cameras="{ front: {type: opencv, index_or_path: 0, width: 640, height: 480, fps: 30, fourcc: 'MJPG'}}" \
    --teleop.type=so101_leader \
    --teleop.port=/dev/ttyCH343USB0 \
    --teleop.id=kang_leader_arm \
    --display_data=true

# ---------------------------------------------------------

# 6. 录制训练数据 (Recording for Training)
# 录制 5 集数据，任务为 "Grab the black ball"
# 注意：此处摄像头配置使用了 'wrist' (手腕) 而非 'side'
# 数据会自动上传到 HuggingFace: mark-two/test
# ---------------------------------------------------------
lerobot-record \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM0 \
    --robot.id=kang_follower_arm \
    --robot.cameras="{ front: {type: opencv, index_or_path: 0, width: 640, height: 480, fps: 30, fourcc: 'MJPG'}, wrist: {type: opencv, index_or_path: 2, width: 640, height: 480, fps: 30, fourcc: 'MJPG'}}" \
    --teleop.type=so101_leader \
    --teleop.port=/dev/ttyACM1 \
    --teleop.id=kang_leader_arm \
    --display_data=true \
    --dataset.repo_id=mark-two/test \    #注意要改名字
    --dataset.num_episodes=5 \
    --dataset.single_task="Grab the black ball" \
    --dataset.push_to_hub=true \
    --dataset.episode_time_s=10 \
    --dataset.reset_time_s=5 


    #手动上传数据
    huggingface-cli upload --repo-type dataset mark-two/testpart2 ~/.cache/huggingface/lerobot/mark-two/testpart2 .

  --steps=300000 \

  lerobot-train \
